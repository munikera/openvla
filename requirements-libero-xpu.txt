# ──────────────────────────────────────────────────────────────────────────────
# requirements-libero-xpu.txt
# Full dependency list to run OpenVLA LIBERO evaluation on Intel Arc XPU.
#
# Install order matters — follow the sections in sequence:
#
#   pip install -r requirements-libero-xpu.txt
#
# Then run with:
#   MUJOCO_GL=osmesa PYOPENGL_PLATFORM=osmesa \
#   PYTHONPATH=/path/to/openvla \
#   python experiments/robot/libero/run_libero_eval.py \
#       --model_family openvla \
#       --pretrained_checkpoint openvla/openvla-7b-finetuned-libero-spatial \
#       --task_suite_name libero_spatial \
#       --center_crop True \
#       --num_trials_per_task 5
# ──────────────────────────────────────────────────────────────────────────────

# ── 1. PyTorch XPU (Intel Arc) ────────────────────────────────────────────────
# Must use XPU wheels — standard PyPI torch has no XPU backend
--extra-index-url https://download.pytorch.org/whl/xpu
torch==2.10.0+xpu
torchvision==0.25.0+xpu
torchaudio==2.10.0+xpu

# ── 2. Core ML / HuggingFace ──────────────────────────────────────────────────
transformers==4.40.1        # Pinned: >4.40.1 breaks attention_mask in cached generate() on XPU
accelerate==1.12.0
huggingface_hub==0.36.2
timm==0.9.10                # Vision backbone (DINOv2/SigLIP); newer versions break compatibility
sentencepiece==0.1.99       # Tokenizer dependency
tokenizers==0.19.1
peft==0.11.1                # LoRA fine-tuning support

# ── 3. Image processing ───────────────────────────────────────────────────────
pillow
tensorflow-cpu              # Used ONLY for image preprocessing (tf.image.resize with lanczos3)
                            # matching training pipeline — no XPU/GPU needed for this

# ── 4. Scientific / utility ───────────────────────────────────────────────────
numpy
einops
matplotlib
imageio[ffmpeg]             # Saving rollout videos

# ── 5. Config / logging ───────────────────────────────────────────────────────
draccus                     # CLI config decorator used by run_libero_eval.py
wandb                       # Optional: pass --use_wandb True to enable
tqdm
rich

# ── 6. LIBERO simulation stack ────────────────────────────────────────────────
# Install robosuite WITHOUT deps first to avoid pulling broken egl_probe
# (egl_probe requires CMake and has an outdated CMakeLists.txt incompatible
#  with CMake >= 3.5; hf-egl-probe is the fixed replacement and installs fine)
#
# Run manually in this order:
#   pip install robosuite==1.4.0 --no-deps
#   pip install libero --no-deps
#   pip install bddl easydict cloudpickle gym gymnasium

# Listed here for reference (use --no-deps for robosuite and libero):
# robosuite==1.4.0           # Physics sim backend (MuJoCo); needs 1.4.0 not 1.4.1 for LIBERO
# libero                     # Task suite / benchmark
bddl                         # BDDL task description language (libero needs bddl, but version matters)
easydict
cloudpickle
gym
gymnasium>=0.29.0

# ── 7. System libraries (install via apt, not pip) ────────────────────────────
# These must be installed at the OS level before running:
#
#   sudo apt-get install -y libgl1 libglib2.0-0t64 libosmesa6-dev
#
# libgl1          : OpenGL runtime (needed by cv2/robosuite)
# libglib2.0-0t64 : GLib runtime
# libosmesa6-dev  : Software offscreen OpenGL renderer (MUJOCO_GL=osmesa)
#                   Required for headless/no-display server rendering

# ── Notes ─────────────────────────────────────────────────────────────────────
# - bitsandbytes (8-bit/4-bit quantization) is CUDA-only → NOT installed here
# - flash-attn is CUDA-only → NOT installed here
#   openvla_utils.py has been patched to use attn_implementation="eager" on XPU
# - dlimp (RLDS training data loader) is NOT needed for eval
#   prismatic/__init__.py has been patched with lazy imports to avoid loading
#   the training pipeline when only running inference/eval
# - egl_probe build will FAIL (broken CMakeLists.txt) — this is expected and safe
#   hf-egl-probe installs successfully and covers the same functionality
